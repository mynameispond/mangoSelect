Step 1

การติดตั้ง

mangoSelect ใช้งานได้ด้วยไฟล์ CSS, JavaScript และไฟล์ภาษาเท่านั้น ไม่ต้อง build เพิ่ม เริ่มจากโหลดไฟล์ตามด้านล่างก่อน แล้วเลือกใช้ snippet ของแต่ละส่วนตามรูปแบบงาน

ไฟล์ที่ต้องใช้

  • mangoselect.css สำหรับหน้าตา dropdown
  • mangoselect.min.css สำหรับ production เมื่อต้องการไฟล์ CSS ขนาดเล็กกว่า
  • mangoselect.js สำหรับ logic ของ component
  • mangoselect.min.js สำหรับ production เมื่อต้องการไฟล์ขนาดเล็กกว่า
  • i18n/th.js ถ้าต้องการข้อความภาษาไทย
  • demo-api.php ใช้เป็นตัวอย่าง endpoint สำหรับ Ajax
ใช้โค้ดชุดนี้เป็นจุดเริ่มต้น แล้วเลือกตัวอย่างของแต่ละส่วนไปปรับใช้ต่อ

โค้ดเริ่มต้นขั้นต่ำ

การใช้งานตัวเลือกเดียวแบบ Ajax พร้อมโค้ด

ดึงข้อมูลจาก demo-api.php เมื่อเปิด dropdown หรือค้นหา พร้อมตัวอย่าง callback สำหรับติดตามสถานะการโหลดข้อมูล

รูปแบบ JSON response ที่รองรับสำหรับ Ajax

{
	"results": [
		{
			"id": "emp-001",
			"text": "EMP-001 / Ava Smith / Sales",
			"html": "<span class='demo-api-row'><span class='demo-api-row-label'>EMP-001 / Ava Smith / Sales</span><span class='demo-api-chip demo-api-chip--red'>2</span></span>",
			"disabled": false
		},
		{
			"text": "Marketing",
			"html": "<div class='demo-api-group-label'>Marketing</div>",
			"children": [
				{
					"id": "emp-002",
					"text": "EMP-002 / Noah Johnson / Marketing"
				},
				{
					"id": "emp-003",
					"html": "<span class='demo-api-row'><span class='demo-api-row-label'>EMP-003 / Emma Wilson / Marketing</span><span class='demo-api-chip demo-api-chip--blue'>4</span></span>"
				}
			]
		}
	],
	"totals": 120
}
  • results คือรายการตัวเลือกที่จะแสดง
  • id คือค่า value
  • text ควรเป็น plain text สำหรับใช้กับ summary, search และ get_selected_texts()
  • html เป็น optional field สำหรับ fallback render และจะถูกใช้ก็ต่อเมื่อ allow_html: true และ renderer ไม่คืนค่า
  • ถ้าส่ง html มาอย่างเดียวและไม่ส่ง text ไลบรารีจะพยายามดึง plain text จาก HTML ให้เอง
  • children ใช้สำหรับสร้าง optgroup ในผลลัพธ์ Ajax
  • disabled เป็น optional field ถ้าไม่ส่งมาจะถือเป็น false
  • totals ใช้ช่วยคำนวณว่ามีหน้าถัดไปสำหรับ infinite scroll หรือไม่
  • ถ้าต้องการ custom UI แบบปลอดภัยกว่า string HTML ให้ใช้ render_option() และ render_group()
ค่าเริ่มต้นของ allow_html คือ false ดังนั้น field html และ string HTML ที่คืนจาก renderer จะไม่ถูก inject เข้า DOM จนกว่าจะเปิดเอง

การปรับแต่งขาเข้าและขาออกสำหรับ API อื่นๆ (Transformer Hooks)

หากหลังบ้าน (Backend API) ของคุณมีโครงสร้างข้อมูลที่ไม่ตรงกับความต้องการเริ่มต้นของไลบรารี คุณสามารถใช้ฟังก์ชันแปลงข้อมูลขาเข้า (Request) และขาออก (Response) เพื่อจับคู่อย่างอิสระโดยไม่ต้องเปลี่ยนโค้ดฝั่ง Server:

1. ฟังก์ชันแปลงข้อมูลฝั่งขาเข้า: transform_request(params)

แปลงพารามิเตอร์ของไลบรารี (ค่าเริ่มต้นส่ง search, page_num, per_page) ไปเป็นตัวแปรที่ระบบหลังบ้านรองรับ หรือเขียนสคริปต์ส่ง Body เป็น JSON String สำหรับ POST Request:

window.mangoSelect.init({
	selector: "#guide-ajax-single",
	ajax: {
		url: "https://api.example.com/users",
		method: "GET",
		// ตัวอย่าง: เปลี่ยนชื่อตัวแปร Query String
		transform_request: function (params) {
			return {
				q: params.search,       // เปลี่ยน search -> q
				page: params.page_num,  // เปลี่ยน page_num -> page
				limit: params.per_page  // เปลี่ยน per_page -> limit
			};
		}
	}
});
// ตัวอย่าง: สำหรับ POST Request ที่ต้องส่งข้อมูลเป็น JSON Body
window.mangoSelect.init({
	selector: "#guide-ajax-single",
	ajax: {
		url: "https://api.example.com/users",
		method: "POST",
		transform_request: function (params) {
			return JSON.stringify({
				searchTerm: params.search,
				pageIndex: params.page_num - 1, // ปรับฐาน 1-indexed เป็น 0-indexed
				pageSize: params.per_page
			});
		},
		headers: {
			"Content-Type": "application/json" // ระบุ Content-Type เป็น JSON
		}
	}
});

2. ฟังก์ชันแปลงข้อมูลฝั่งขาออก: process_results(payload, params)

แปลงรูปแบบข้อมูลที่ฝั่ง Server ตอบกลับมา (Response Payload) ให้อยู่ในโครงสร้างอาเรย์หรือรูปแบบที่ไลบรารีเข้าใจ:

window.mangoSelect.init({
	selector: "#guide-ajax-single",
	ajax: {
		url: "https://api.example.com/users",
		// ตัวอย่าง: โครงสร้าง API เดิมส่ง { status: "success", data: { list: [...] } }
		process_results: function (payload, params) {
			return {
				results: payload.data.list.map(function (item) {
					return {
						id: item.user_id,
						text: item.full_name,
						description: item.department_name
					};
				}),
				// ช่วยคำนวณว่ามีหน้าถัดไปสำหรับ Infinite Scroll หรือไม่
				pagination: {
					more: payload.data.list.length >= params.per_page
				}
			};
		}
	}
});

attr data-mangoselect-* ที่ใช้ได้จริง

ตารางนี้สรุป attributes ที่ใช้บ่อยและควรใช้ในงานจริง ส่วน option อื่นในตาราง JavaScript options สามารถตั้งผ่าน data-mangoselect-<option_name> ได้ถ้าใช้ชื่อตรงกับ option

Attribute ใช้กับ คืออะไร ตัวอย่าง
data-mangoselect-language Local / Ajax กำหนดภาษาของข้อความใน component เช่น placeholder, search, clear data-mangoselect-language="th"
data-mangoselect-placeholder Local / Ajax ข้อความ placeholder ที่แสดงก่อนผู้ใช้เลือกค่า data-mangoselect-placeholder="เลือกจังหวัด"
data-mangoselect-close Local / Ajax แสดงปุ่ม Close ภายใน dropdown data-mangoselect-close="true"
data-mangoselect-allow_html Local / Ajax อนุญาตให้ render field html และ string HTML ที่คืนจาก renderer ด้วย innerHTML data-mangoselect-allow_html="true"
data-mangoselect-min_selected Multiple จำนวนขั้นต่ำที่ต้องคงไว้ ถ้าเลือกถึงขั้นต่ำแล้วจะลบออกต่อไม่ได้ data-mangoselect-min_selected="1"
data-mangoselect-max_selected Multiple จำนวนสูงสุดที่เลือกได้ ถ้าครบแล้ว option อื่นจะถูกปิดไม่ให้เลือกเพิ่ม data-mangoselect-max_selected="3"

data-mangoselect-per_page Ajax กำหนดจำนวนรายการต่อ request data-mangoselect-per_page="20"
data-mangoselect-delay Ajax กำหนดเวลาหน่วงก่อนยิง request ตอนค้นหา หน่วยเป็นมิลลิวินาที data-mangoselect-delay="300"
data-mangoselect-search_length Ajax จำนวนตัวอักษรขั้นต่ำที่ต้องพิมพ์ก่อนเริ่มยิง Ajax request data-mangoselect-search_length="2"
data-mangoselect-render_option Local / Ajax ชื่อ global function สำหรับ custom render ต่อหนึ่ง option data-mangoselect-render_option="demoRenderOptionHtml"
data-mangoselect-render_group Local / Ajax Group ชื่อ global function สำหรับ custom render หัวข้อ group data-mangoselect-render_group="demoRenderGroupHtml"
data-mangoselect-render_checkbox Multiple ชื่อ global function สำหรับ custom render checkbox ต่อหนึ่ง option ถ้าไม่ตั้งค่าจะใช้ checkbox ธรรมดา data-mangoselect-render_checkbox="demoRenderCheckboxHtml"
data-mangoselect-param-* Ajax ทุก attribute ที่ขึ้นต้นด้วย data-mangoselect-param- จะถูกแปลงเป็น query param อัตโนมัติ ส่วน config หลักต้องใช้ชื่อ option ตรงตัว เช่น min_selected, max_selected, search_length, close data-mangoselect-param-branch="bkk"
data-mangoselect-param-customer="1"

ทุก option ใน init() ตั้งผ่าน data-mangoselect-<option_name> ได้ถ้าใช้ชื่อให้ตรงกับ option และลำดับการใช้ค่าคือ attr -> option -> default ส่วน function options ที่ตั้งผ่าน attr ต้องชี้ไปที่ global function บน window

JavaScript options ที่รองรับทั้งหมด

ตารางนี้สรุป option หลักที่ใช้กับ window.mangoSelect.init() โดยตรง พร้อมรูปแบบค่าที่ควรกำหนดและผลของแต่ละตัวเลือก ส่วน callback และ function options อยู่ในหัวข้อถัดไป

Option ค่าเริ่มต้น กำหนดแบบไหน มีผลยังไง ตัวอย่าง
selector "" string selector ของ <select> ต้องระบุเองทุกครั้ง โดยในเอกสารนี้ใช้แบบ id เป็นหลัก และถ้าต้องการ init หลายตัวจาก class ให้ใช้ selector: ".user-picker" โดยตรง ถ้าไม่ส่งค่า ไลบรารีจะไม่ init selector: "#guide-tags"
placeholder "" string ข้อความที่แสดงบน trigger ก่อนมีการเลือกค่า ถ้ากำหนดทั้ง JS และ attr ค่าใน JavaScript จะถูกใช้ก่อน placeholder: "เลือกพนักงาน"
summary_limit 2 number ใน multiple select ถ้าเลือกเกินจำนวนนี้ trigger จะสรุปเป็นจำนวนรายการแทนการแสดงชื่อทั้งหมด summary_limit: 3
language "en" string หรือ object ภาษา ใช้เปลี่ยนข้อความใน component ทั้งระบบ เช่น placeholder, search, clear, loading และข้อความสรุป language: "th"
search false boolean เปิดหรือปิดช่องค้นหาใน dropdown ถ้าเปิด tags ไลบรารีจะบังคับให้มี search ให้อัตโนมัติ search: true
search_length null number ใช้กับ Ajax เท่านั้น ถ้ากำหนด 2 จะเริ่มยิง request เมื่อพิมพ์ครบ 2 ตัวอักษร ส่วน local search จะไม่สนใจ option นี้ search_length: 2
select_all false boolean แสดงปุ่มเลือกทั้งหมดด้านล่าง dropdown ใน multiple select ถ้าปิดไว้ปุ่มจะไม่ถูก render select_all: true
clear_all false boolean แสดงปุ่มล้างค่าทั้งหมด หรือปุ่มล้างค่าใน single select clear_all: true
close false boolean แสดงปุ่ม Close เพิ่มใน action bar ของ dropdown close: true
close_after_select_all false boolean หลังจากกด Select all และเขียนค่าลง select สำเร็จ จะสั่งปิด dropdown ให้อัตโนมัติ ถ้าเปิด ok_cancel_in_multi จะรอให้กด OK ก่อน close_after_select_all: true
close_after_clear_all false boolean หลังจากกด Clear all แล้วค่าถูกล้างสำเร็จ จะสั่งปิด dropdown ให้อัตโนมัติ ถ้าอยู่ใน draft mode จะรอการยืนยันก่อน close_after_clear_all: true
ok_cancel_in_multi false boolean ใช้กับ multiple select เพื่อเปลี่ยนเป็นโหมด draft ค่าใน <select> จริงจะยังไม่ถูกเปลี่ยนจนกว่าจะกด OK ok_cancel_in_multi: true
tags false boolean เพิ่มปุ่ม + ใน search box เพื่อสร้าง option ใหม่จากข้อความที่พิมพ์ ถ้าค่านั้นมีอยู่แล้วจะเลือก option เดิมแทน รองรับทั้ง local, ajax, single select และ multiple select tags: true
allow_html false boolean อนุญาตให้ render field html และ string HTML ที่คืนจาก renderer ด้วย innerHTML ถ้า false จะ render string จาก renderer เป็น plain text allow_html: true
min_selected 0 number กำหนดจำนวนขั้นต่ำที่ต้องคงไว้ใน multiple select ถ้าถึงขั้นต่ำแล้วจะยกเลิกเลือกต่อไม่ได้ min_selected: 1
max_selected null number หรือ null กำหนดจำนวนสูงสุดที่เลือกได้ ถ้าเลือกครบแล้ว option อื่นจะถูกปิดไม่ให้เลือกเพิ่ม max_selected: 3
ajax null true, string URL หรือ object เปิด remote mode โดยใช้ URL จาก attr หรือส่ง config เพิ่ม เช่น method, headers, data, process_results, search_length รายละเอียดดูตัวอย่างด้านล่าง ajax: true

ตัวอย่างการกำหนด ajax แบบ object

กำหนด ajax ได้ 3 แบบ คือ true เพื่อใช้ URL จาก data-mangoselect-url, กำหนดเป็น string URL โดยตรง หรือกำหนดเป็น object เพื่อควบคุม request และการแปลง response เพิ่มเติม 

window.mangoSelect.init({
	selector: "#employee-select",
	language: "th",
	ajax: {
		url: "/api/employee/search",
		method: "POST",
		headers: {
			"X-CSRF-TOKEN": window.csrfToken,
			"Authorization": "Bearer YOUR_TOKEN"
		},
		data: function (baseParams) {
			return {
				customer_id: 7,
				branch_code: "bkk",
				keyword: baseParams.search
			};
		},
		process_results: function (payload, params) {
			return {
				results: payload.items || [],
				totals: payload.total_rows || 0
			};
		}
	}
});
  • url ใช้ได้ทั้ง string และ function เช่น url: function (params) { return "/api/employee/search"; }
  • method ถ้าเป็น GET จะส่งค่าเป็น query string แต่ถ้าไม่ใช่ GET ไลบรารีจะส่งเป็น application/x-www-form-urlencoded
  • headers กำหนดเป็น object ของ header ที่ต้องส่งเพิ่ม เช่น token หรือ authorization
  • data ใช้ได้ทั้ง object และ function โดยค่าที่ return จะถูกรวมกับพารามิเตอร์หลักของไลบรารี
  • พารามิเตอร์หลัก search, page_num และ per_page จะถูกส่งเสมอ แม้จะกำหนด data เพิ่มเอง
  • process_results ใช้แปลงรูปแบบ response ของ backend ให้กลายเป็น { results, totals } ก่อน render
  • ไลบรารีจะใส่ header X-Requested-With: XMLHttpRequest ให้อัตโนมัติทุก request
ถ้า backend รับชื่อ field ไม่ตรงกับ canonical params ของไลบรารี เช่นต้องการ keyword แทน search ให้เพิ่ม field ใหม่ผ่าน data ได้ แต่ search, page_num และ per_page จะยังถูกส่งไปด้วยเสมอ

callback และ function options

รองรับ callback สำหรับ lifecycle ของ dropdown, การเปลี่ยนค่า และสถานะของ Ajax request รวมถึง function options ใน ajax และ language config ที่ใช้ปรับพฤติกรรมเพิ่มเติม

ชื่อ กำหนดตรงไหน ทำงานเมื่อไร รับค่า / คืนค่า ตัวอย่าง
on_open(detail) options หลักของ mangoSelect.init() เมื่อ dropdown ถูกเปิดผ่าน trigger หรือผ่าน API รับ detail ที่มี open_reason, selected_values, is_open และข้อมูลของ instance on_open: function (detail) { console.log(detail.open_reason); }
on_close(detail) options หลักของ mangoSelect.init() เมื่อ dropdown ถูกปิดทุกกรณี รวมถึงปิดผ่าน API, ปิดเพราะ disabled หรือถูก destroy() รับ detail ที่มี close_reason, has_changed, selected_values และ last_change_detail on_close: function (detail) { console.log(detail.close_reason); }
on_change(detail) options หลักของ mangoSelect.init() ทุกครั้งที่ค่าถูกเลือก, ยกเลิกเลือก, clear, set ผ่าน API หรือเปลี่ยนจาก remote result รับ detail ที่มี action, changed_value, selected_values, selected_texts, instance และอื่น ๆ on_change: function (detail) { console.log(detail.selected_values); }
on_change_all(detail) options หลักของ mangoSelect.init() ทำงานตอน dropdown ถูกปิดเช่นเดียวกับ on_close เหมาะกับกรณีที่ต้องการอ่านค่ารวมหลังปิด dropdown รับ detail ที่มีค่าปัจจุบันทั้งหมด พร้อม opened_selected_values, has_changed, close_reason และ last_change_detail on_change_all: function (detail) { console.log(detail.close_reason, detail.selected_values); }
on_loading_start(detail) options หลักของ mangoSelect.init() ก่อนยิง request Ajax ของ instance นั้น รับ detail ที่มี request_params, request_url, request_method, page_num และ should_reset on_loading_start: function (detail) { console.log(detail.request_params.search); }
on_loading_end(detail) options หลักของ mangoSelect.init() เมื่อ request Ajax จบ ทั้งกรณี success และ error รับ detail ที่มี success, status, results, totals, has_more และข้อมูล request เดิม on_loading_end: function (detail) { console.log(detail.success, detail.status); }
on_error(detail) options หลักของ mangoSelect.init() เมื่อ request Ajax ล้มเหลว, network error, HTTP status ไม่ผ่าน หรือ parse JSON ไม่ได้ รับ detail ที่มี error_type, error_message, status และ response_text on_error: function (detail) { console.log(detail.error_type, detail.error_message); }
render_option(option) options หลักของ mangoSelect.init() หรือ data-mangoselect-render_option custom render ต่อหนึ่ง option โดยคืน Node, string หรือปล่อยให้ fallback ไปใช้ html / text ถ้า allow_html: true และคืน string จะ render เป็น HTML รับ option ที่มี id, value, text, html, disabled, selected และ is_selected render_option: function (option) { return '<div class="item">' + option.text + '</div>'; }
render_group(group) options หลักของ mangoSelect.init() หรือ data-mangoselect-render_group custom render หัวข้อ group / optgroup title ถ้า allow_html: true และคืน string จะ render เป็น HTML รับ group ที่มี text, label, html, disabled, group_key และ children render_group: function (group) { return '<div class="group">' + group.text + '</div>'; }
render_checkbox(option) options หลักของ mangoSelect.init() หรือ data-mangoselect-render_checkbox custom render checkbox ใน multiple select โดยคืน Node หรือ string ถ้าเปิด allow_html: true string จะ render เป็น HTML ถ้าไม่ตั้งค่าจะใช้ checkbox ธรรมดา รับ option ชุดเดียวกับ render_option() และใช้ option.is_selected เพื่อแยกหน้าตาเลือก/ไม่เลือก render_checkbox: function (option) { return option.is_selected ? "<span>on</span>" : "<span>off</span>"; }
ajax.url(params) ajax.url ก่อนยิง request ใช้ในกรณีที่ URL ต้องคำนวณแบบ dynamic รับพารามิเตอร์ request ปัจจุบัน และต้องคืนค่า URL เป็น string url: function (params) { return "/api/users?customer=" + params.customer; }
ajax.data(params) ajax.data ก่อนส่ง request เพื่อเพิ่มหรือตกแต่ง query params รับ params และควรคืน object ที่ต้องการ merge เพิ่มเข้าไป data: function (params) { params.status = "active"; return params; }
ajax.process_results(payload, params) ajax.process_results หลังจากได้ response จาก server และก่อนแปลงเป็นรายการตัวเลือก รับข้อมูลดิบจาก server แล้วคืน payload ใหม่ที่มี results หรือ data process_results: function (payload) { return { results: payload.items, totals: payload.total }; }
language.selected_count(args) ภาษา custom หรือ register_language() ตอนเลือกหลายค่าเกิน summary_limit และต้องแสดงข้อความสรุปจำนวน รับ { count: number } แล้วคืนข้อความที่ต้องการแสดง selected_count: function (args) { return "เลือกแล้ว " + args.count + " รายการ"; }

การใช้ destroy()

ใช้เมื่อไม่ต้องการ instance เดิมแล้ว เช่น ปิด modal, เปลี่ยนเนื้อหาเดิมออก หรือเตรียม init ใหม่ โดย destroy() จะยกเลิก timer/request ของ instance และถ้าไม่มี instance เหลือแล้วจะถอด global document listeners ให้อัตโนมัติ 

var instance = window.mangoSelect.get_instance("#guide-ajax-single");

if (instance) {
	instance.destroy();
}

var newInstance = window.mangoSelect.init({
	selector: "#guide-ajax-single",
	language: "th"
});
